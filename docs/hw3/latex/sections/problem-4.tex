\section{فضای آدرس ریسه‌ها و درهم‌آمیزی دستورها}

\subsection*{بخش آ – آدرس‌های چاپ‌شده در ریسه‌ها}

ریسه‌های یک پردازه کد، داده‌ی سراسری و هیپ را مشترک دارند، ولی هر ریسه پشته‌ی خودش را دارد.
\begin{center}\small\atlrows
\begin{tabular}{@{}c>{\centering\arraybackslash}p{1.8cm}>{\raggedleft\arraybackslash}p{8cm}@{}}
  \rowcolor{atl-primary}\atlhead{خط} & \atlhead{یکسان؟} & \atlhead{دلیل}\\
  \code{\&foo} & بله & کد مشترک است\\
  \code{\&global} & بله & داده‌ی سراسری مشترک است\\
  \code{\&arg} & خیر & آدرس پارامتر روی پشته‌ی همان ریسه است و پشته‌ها جدا هستند\\
  \code{arg} & بله & مقدار پارامتر است که برای هر سه ریسه همان \code{hmem} (اشاره‌گر هیپ) است\\
\end{tabular}
\end{center}
پس خروجی خط‌های ۱، ۲ و ۴ در همه‌ی ریسه‌ها یکی است و فقط خط ۳ فرق می‌کند.

\subsection*{بخش ب – درهم‌آمیزی اجرای دو دستور}

هر ریسه \lr{$S_1$} (\code{x = x + 1}) و سپس \lr{$S_2$} (\code{y = x + y}) را اجرا می‌کند؛ هر دو اتمیک‌اند. هر ریسه \lr{$S_1$} را یک بار اجرا می‌کند، پس \lr{$x$} نهایی همیشه \lr{$n+1$} است.

\lr{$S_2$}ها را به ترتیب زمان شماره بزنید و \lr{$x_k$} را مقدار \lr{$x$} در \lr{$S_2$}ی \lr{$k$}ام بگیرید. پیش از آن دست‌کم \lr{$k$} ریسه \lr{$S_1$} را اجرا کرده‌اند (\lr{$k-1$} ریسه‌ی قبلی و خودش) و حداکثر همه‌ی \lr{$n$} ریسه، پس
\begin{center}
\mbox{$y=1+\sum_{k=1}^{n}x_k,\qquad x_1\le x_2\le\cdots\le x_n,\qquad k+1\le x_k\le n+1$}
\end{center}
و هر دنباله‌ی مجاز با یک زمان‌بندی ساخته می‌شود.

\begin{enumerate}
  \item \textbf{۲ ریسه:} \lr{$x=3$}؛ \lr{$(x_1,x_2)$} برابر \lr{$(2,3)$} یا \lr{$(3,3)$} است، پس \lr{$y\in\{6,7\}$}.
  \item \textbf{۳ ریسه:} \lr{$x=4$}؛ پنج دنباله‌ی مجاز \lr{$(x_1,x_2,x_3)$} داریم و \lr{$y\in\{10,11,12,13\}$} است:
  \begin{center}\mbox{$(2,3,4),\ (2,4,4),\ (3,3,4),\ (3,4,4),\ (4,4,4)\ \Rightarrow\ y=10,\ 11,\ 11,\ 12,\ 13$}\end{center}
  \item \textbf{بیشینه‌ی \lr{$y$}:} همه‌ی \lr{$S_1$}ها پیش از همه‌ی \lr{$S_2$}ها؛ هر \lr{$x_k=n+1$} و
  \mbox{$y_{\max}=1+n(n+1)=n^2+n+1$}.
  \item \textbf{کمینه‌ی \lr{$y$}:} ریسه‌ها پشت‌سرهم اجرا شوند؛ \lr{$x_k=k+1$} و
  \mbox{$y_{\min}=1+\sum_{k=1}^{n}(k+1)=\frac{(n+1)(n+2)}{2}$}.
\end{enumerate}
