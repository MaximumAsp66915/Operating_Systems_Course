% سبک‌های مشترک شکل‌ها. شکل‌ها باید داخل \begin{LTR} باشند؛
% متن فارسی داخل گره‌ها با \beginR/\endR راست‌به‌چپ می‌شود و هر نوشته‌ی لاتین باید در \lr باشد.
\tikzset{
  pnode/.style={draw=atl-primary, fill=atl-tint, rounded corners=3pt, minimum height=8mm,
    minimum width=13mm, inner sep=3pt, align=center, line width=0.6pt,
    execute at begin node=\beginR, execute at end node=\endR},
  pnodez/.style={pnode, draw=atl-accent, fill=atl-accent!22},
  pedge/.style={->, >=Stealth, line width=0.7pt, color=atl-primary},
  elab/.style={font=\scriptsize, inner sep=1.5pt, fill=white, align=center,
    execute at begin node=\beginR, execute at end node=\endR},
  tlab/.style={font=\scriptsize, inner sep=1pt, align=center,
    execute at begin node=\beginR, execute at end node=\endR},
}
